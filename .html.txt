  <script>
    const myId = 'Vendég #' + Math.floor(100 + Math.random() * 900);
    let trust = 860;
    let partnerId = 'Ismeretlen #' + Math.floor(100 + Math.random() * 900);
    let isCrossTab = false;

    document.getElementById('self-id').textContent = myId;
    document.getElementById('partner-label').textContent = 'Partner: ' + partnerId;

    // 1. Saját kamera indítása (vagy canvas fallback, ha nincs engedély)
    const localVideo = document.getElementById('local-video');
    const localFallback = document.getElementById('local-fallback');
    navigator.mediaDevices?.getUserMedia({ video: true, audio: true })
      .then(stream => { localVideo.srcObject = stream; })
      .catch(() => {
        // Fallback saját avatár animáció, ha nincs webkamera vagy letiltott
        localVideo.style.display = 'none';
        localFallback.style.display = 'block';
        drawSelfAvatar(localFallback);
      });

    // 2. Animált, élő távoli partner canvas (nem egy állókép!)
    const rCanvas = document.getElementById('remote-canvas');
    const rCtx = rCanvas.getContext('2d');
    let frame = 0;
    function renderRemotePartner() {
      rCanvas.width = rCanvas.clientWidth || 400;
      rCanvas.height = rCanvas.clientHeight || 300;
      const w = rCanvas.width;
      const h = rCanvas.height;

      // Háttér gradiens
      const grad = rCtx.createLinearGradient(0, 0, w, h);
      grad.addColorStop(0, '#1c1b3a');
      grad.addColorStop(1, '#0c1322');
      rCtx.fillStyle = grad;
      rCtx.fillRect(0, 0, w, h);

      // Finom mozgó fej és szemek szimulációja
      frame++;
      const swayX = Math.sin(frame * 0.03) * 6;
      const swayY = Math.cos(frame * 0.04) * 4;
      const centerX = w / 2 + swayX;
      const centerY = h / 2 + 20 + swayY;

      // Vállak és test
      rCtx.fillStyle = '#222842';
      rCtx.beginPath();
      rCtx.ellipse(centerX, centerY + 130, 90, 70, 0, 0, Math.PI * 2);
      rCtx.fill();

      // Arc
      rCtx.fillStyle = '#f6b89e';
      rCtx.beginPath();
      rCtx.ellipse(centerX, centerY, 48, 62, 0, 0, Math.PI * 2);
      rCtx.fill();

      // Haj
      rCtx.fillStyle = '#3a2518';
      rCtx.beginPath();
      rCtx.arc(centerX, centerY - 25, 52, Math.PI, 0, false);
      rCtx.fill();

      // Pislogó szemek
      const isBlinking = (frame % 120 > 114);
      rCtx.fillStyle = '#1c1c1c';
      if (isBlinking) {
        rCtx.fillRect(centerX - 24, centerY - 6, 14, 2);
        rCtx.fillRect(centerX + 10, centerY - 6, 14, 2);
      } else {
        rCtx.beginPath();
        rCtx.arc(centerX - 17, centerY - 5, 4, 0, Math.PI * 2);
        rCtx.arc(centerX + 17, centerY - 5, 4, 0, Math.PI * 2);
        rCtx.fill();
      }

      // Beszédmozgás
      const mouthOpen = Math.abs(Math.sin(frame * 0.08)) * 6;
      rCtx.fillStyle = '#a64d4d';
      rCtx.beginPath();
      rCtx.ellipse(centerX, centerY + 30, 10, 2 + mouthOpen, 0, 0, Math.PI * 2);
      rCtx.fill();

      requestAnimationFrame(renderRemotePartner);
    }
    renderRemotePartner();

    function drawSelfAvatar(canv) {
      const ctx = canv.getContext('2d');
      function loop() {
        canv.width = canv.clientWidth || 300;
        canv.height = canv.clientHeight || 200;
        ctx.fillStyle = '#0f1d33';
        ctx.fillRect(0, 0, canv.width, canv.height);
        ctx.fillStyle = '#7c3aed';
        ctx.beginPath();
        ctx.arc(canv.width / 2, canv.height / 2, 40, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#fff';
        ctx.font = '14px Outfit';
        ctx.textAlign = 'center';
        ctx.fillText('Saját kamera aktív', canv.width / 2, canv.height / 2 + 65);
        requestAnimationFrame(loop);
      }
      loop();
    }

    // 3. Valós több-ablakos kapcsolat (BroadcastChannel)
    const channel = new BroadcastChannel('kapcs_channel');
    channel.onmessage = (e) => {
      const { type, sender, text } = e.data;
      if (sender === myId) return;

      if (type === 'hello') {
        isCrossTab = true;
        partnerId = sender;
        document.getElementById('partner-label').textContent = 'Partner (Élő lap): ' + partnerId;
        addMsg('Egy másik böngészőfül csatlakozott a szobához!', 'sys');
      }
      if (type === 'chat') {
        addMsg(text, 'incoming');
      }
      if (type === 'next') {
        addMsg('A partnert továbblépett.', 'sys');
        partnerId = 'Ismeretlen #' + Math.floor(100 + Math.random() * 900);
        document.getElementById('partner-label').textContent = 'Partner: ' + partnerId;
      }
    };
    channel.postMessage({ type: 'hello', sender: myId });

    // 4. Chat funkció
    const chatForm = document.getElementById('chat-form');
    const chatIn = document.getElementById('chat-in');
    const chatBody = document.getElementById('chat-body');

    chatForm.addEventListener('submit', (e) => {
      e.preventDefault();
      const txt = chatIn.value.trim();
      if (!txt) return;
      addMsg(txt, 'outgoing');
      chatIn.value = '';

      if (isCrossTab) {
        channel.postMessage({ type: 'chat', sender: myId, text: txt });
      } else {
        // Automata intelligens válasz, ha 1 lapon teszteled
        setTimeout(() => {
          const answers = ['Szia! Hallak tisztán.', 'Hogy telik a napod?', 'Örülök a találkozásnak!', 'Itt vagyok, működik a kapcsolat.'];
          addMsg(answers[Math.floor(Math.random() * answers.length)], 'incoming');
        }, 800);
      }
    });

    function addMsg(text, type) {
      const el = document.createElement('div');
      el.className = 'msg ' + type;
      el.textContent = text;
      chatBody.appendChild(el);
      chatBody.scrollTop = chatBody.scrollHeight;
    }

    // 5. GuardAI tesztelő és homályosító pajzs
    const shield = document.getElementById('guard-shield');
    document.getElementById('btn-guardai').addEventListener('click', () => {
      shield.classList.add('active');
      addMsg('GuardAI: Képi elemzés nem megfelelő tartalmat érzékelt. Videó letiltva.', 'sys');
      // Admin táblázat frissítése
      const tr = document.createElement('tr');
      tr.innerHTML = `<td><span class="pill high">MAGAS</span></td><td>GuardAI képvédelmi trigger</td><td>${partnerId}</td><td><button style="border:0; background:rgba(239,68,68,0.2); color:#fca5a5; padding:4px 8px; border-radius:4px; cursor:pointer;" onclick="this.closest('tr').remove(); updateBadge();">Kitiltás</button></td>`;
      document.getElementById('admin-tbody').prepend(tr);
      updateBadge();
    });

    document.getElementById('btn-unblur').addEventListener('click', () => shield.classList.remove('active'));

    // 6. Következő partner váltása & Modálok
    document.getElementById('btn-next').addEventListener('click', () => {
      shield.classList.remove('active');
      channel.postMessage({ type: 'next', sender: myId });
      document.getElementById('modal-rate').classList.add('open');
    });

    document.getElementById('btn-report').addEventListener('click', () => {
      document.getElementById('modal-report').classList.add('open');
    });

    document.getElementById('btn-submit-report').addEventListener('click', () => {
      closeModals();
      shield.classList.add('active');
      addMsg('A partnert bejelentetted és leválasztottad.', 'sys');
      partnerId = 'Ismeretlen #' + Math.floor(100 + Math.random() * 900);
      document.getElementById('partner-label').textContent = 'Partner: ' + partnerId;
    });

    function addTrust(points) {
      trust += points;
      document.getElementById('trust-val').textContent = trust;
      document.getElementById('self-trust').textContent = `Bizalmi pont: ${trust}`;
      closeModals();
      addMsg('Köszönjük az értékelést! Partner frissítve.', 'sys');
      partnerId = 'Ismeretlen #' + Math.floor(100 + Math.random() * 900);

      document.querySelectorAll('.modal-wrap').forEach(m => m.classList.remove('open'));

      const count = document.getElementById('admin-tbody').children.length;
      document.getElementById('report-badge').textContent = count;
    }

      camOn = !camOn;
      this.classList.toggle('off', !camOn);
      if (localVideo.srcObject) localVideo.srcObject.getVideoTracks().forEach(t => t.enabled = camOn);
    });
    document.getElementById('btn-mic').addEventListener('click', function() {
      micOn = !micOn;
      this.classList.toggle('off', !micOn);
      if (localVideo.srcObject) localVideo.srcObject.getAudioTracks().forEach(t => t.enabled = micOn);
    });
